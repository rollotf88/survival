# .htaccess du jeu ZOMBRIA chez o2switch (copié sous le nom « .htaccess » dans site/ par preparer-site.py)
# 1) Toujours en https et toujours sous le nom zombria.com (le sous-domaine technique créé par cPanel,
#    ou www.zombria.com, renvoient vers https://zombria.com) : le jeu ne s'affiche jamais sous un autre nom.
# 2) /api/… est laissé au serveur Node.js (application « Setup Node.js App » du cPanel, adresse zombria.com/api).
# 3) Le fichier sw.js (copie hors connexion) n'est jamais gardé en cache, sinon les mises à jour n'arriveraient pas.

RewriteEngine On

# (sauf .well-known : o2switch s'en sert pour renouveler le certificat https)
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTP_HOST} !^zombria\.com$ [NC]
RewriteRule ^(.*)$ https://zombria.com/$1 [R=301,L]
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://zombria.com/$1 [R=301,L]

# 4) Pages, scripts, styles et données : le navigateur doit toujours revérifier (sinon il garde un vieux script
#    qui pointe vers d'anciennes images, même avec F5 ; 2026-11-16). Les images gardent le cache normal :
#    une image redessinée prend toujours un nouveau nom.
<IfModule mod_headers.c>
  <FilesMatch "\.(html|js|css|json|txt|xml)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
</IfModule>

<IfModule mod_headers.c>
  <Files "sw.js">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
  </Files>
</IfModule>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

# Pas de liste des fichiers d'un dossier
Options -Indexes
